function handleAirlineChange() { const airlineInput = document.getElementById("airline-input"); const cabinSelect = document.getElementById("cabin-select"); const summaryAirline = document.getElementById("summary-airline"); const rawInput = airlineInput.value.trim(); const inputVal = rawInput.toLowerCase(); // Clear previous options cabinSelect.innerHTML = ""; summaryAirline.textContent = rawInput || "--"; // If no text typed, disable cabin select if (!rawInput) { cabinSelect.disabled = true; const defaultOption = document.createElement("option"); defaultOption.value = ""; defaultOption.textContent = "Type an airline above first..."; cabinSelect.appendChild(defaultOption); updateTicketPrice(); return; } // 1. Check if the typed airline matches any predefined brand let matchedKey = Object.keys(globalAirlineCabins).find(key => inputVal.includes(key) || key.includes(inputVal) ); let cabinsToLoad = []; if (matchedKey) { cabinsToLoad = globalAirlineCabins[matchedKey]; } else { // 2. Generic cabins for ANY typed airline name const formattedName = rawInput.charAt(0).toUpperCase() + rawInput.slice(1); cabinsToLoad = [ `${formattedName} Economy Class`, `${formattedName} Premium Economy`, `${formattedName} Business Class`, `${formattedName} First Class` ]; } // Unlock select menu and populate options cabinSelect.disabled = false; // Add default "Select Class" option const placeholder = document.createElement("option"); placeholder.value = ""; placeholder.textContent = "-- Select Cabin Class --"; cabinSelect.appendChild(placeholder); cabinsToLoad.forEach(cabin => { const option = document.createElement("option"); option.value = cabin; option.textContent = cabin; cabinSelect.appendChild(option); }); updateTicketPrice(); }
